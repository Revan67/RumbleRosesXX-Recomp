# Optional single-function A/B overlay, generated exclusively from local output.
# No translated game implementation is stored in this handwritten build file.
set(RRXX_FADE_BOUNDARY_INPUT "" CACHE FILEPATH "Isolated regenerated unit containing the fade boundary correction")
if(RRXX_FADE_BOUNDARY_INPUT)
  file(READ "${RRXX_FADE_BOUNDARY_INPUT}" boundary_unit)
  string(REPLACE "\r\n" "\n" boundary_unit "${boundary_unit}")
  string(FIND "${boundary_unit}" "PPC_FUNC_IMPL(__imp__sub_82062200) {" boundary_start)
  if(boundary_start LESS 0)
    message(FATAL_ERROR "Boundary input is missing the selected implementation")
  endif()
  string(SUBSTRING "${boundary_unit}" ${boundary_start} -1 boundary_tail)
  string(FIND "${boundary_tail}" "\n}\n" boundary_end)
  if(boundary_end LESS 0)
    message(FATAL_ERROR "Boundary input has no complete function body")
  endif()
  math(EXPR boundary_length "${boundary_end} + 3")
  string(SUBSTRING "${boundary_tail}" 0 ${boundary_length} boundary_body)
  string(REPLACE "PPC_FUNC_IMPL(__imp__sub_82062200)" "PPC_FUNC_IMPL(sub_82062200)" boundary_body "${boundary_body}")
  set(boundary_output "${CMAKE_CURRENT_BINARY_DIR}/local_fade_boundary.cpp")
  file(GENERATE OUTPUT "${boundary_output}" CONTENT "// Local generated overlay. Never distribute or commit.\n#include \"ppc_recomp_shared.h\"\n${boundary_body}")
  list(APPEND RRXX_TRANSLATED_SOURCES "${boundary_output}")
  if(RRXX_TRANSLATED_MEMORY)
    set_property(SOURCE "${boundary_output}" APPEND PROPERTY COMPILE_OPTIONS "/FI${CMAKE_CURRENT_SOURCE_DIR}/generated_memory.h")
  endif()
endif()

function(rrxx_configure_target target)
  target_compile_definitions(${target} PRIVATE NOMINMAX REX_PLATFORM_WINDOWS=1 SPDLOG_FMT_EXTERNAL)
  target_compile_options(${target} PRIVATE /W0 /bigobj
    "/FI${CMAKE_CURRENT_SOURCE_DIR}/context.h")
  target_include_directories(${target} PRIVATE "${RRXX_PPC_DIR}"
    "${REXSDK_SOURCE}/include" "${REXSDK_SOURCE}/thirdparty/simde"
    "${REXSDK_SOURCE}/thirdparty/simde/simde" "${REXSDK_SOURCE}/thirdparty/fmt/include"
    "${REXSDK_SOURCE}/thirdparty/spdlog/include")
  if(RRXX_NONLOCAL_BOUNDARY_DIR)
    # Transfers cross generated C-linkage functions. Do not assume extern C
    # calls are nonthrowing (/EHsc's implicit /EHc assumption is unsuitable).
    target_compile_options(${target} PRIVATE /EHc-)
    target_compile_definitions(${target} PRIVATE RRXX_NONLOCAL_BRIDGE=1)
    target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  endif()
endfunction()

  COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${REXSDK_RUNTIME_DIR}/rexruntime.dll" "${CMAKE_CURRENT_BINARY_DIR}"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${REXSDK_RUNTIME_DIR}/rexgpu-xenos.dll" "${CMAKE_CURRENT_BINARY_DIR}"
  COMMENT "Synchronizing SDK runtime DLLs")

  target_link_options(${target} PRIVATE /MAP)
